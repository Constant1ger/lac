cmake_minimum_required(VERSION 3.16)
project(lac VERSION 0.0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(lac src/main.c src/lexer.c src/parser.c src/codegen.c)
target_compile_definitions(lac PRIVATE LAC_RUNTIME_PATH="${CMAKE_CURRENT_SOURCE_DIR}/runtime/lac_runtime.h")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lac PRIVATE -Wall -Wextra -Wpedantic)
endif()
