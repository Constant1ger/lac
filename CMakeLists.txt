cmake_minimum_required(VERSION 3.19)
project(lac VERSION 0.0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_path(LIBCLANG_INCLUDE_DIR clang-c/Index.h HINTS C:/msys64/ucrt64/include REQUIRED)
find_library(LIBCLANG_LIBRARY NAMES clang libclang HINTS C:/msys64/ucrt64/lib REQUIRED)
add_executable(lac src/main.c src/lexer.c src/parser.c src/codegen.c src/ffi.c)
target_include_directories(lac PRIVATE ${LIBCLANG_INCLUDE_DIR})
target_link_libraries(lac PRIVATE ${LIBCLANG_LIBRARY})
target_compile_definitions(lac PRIVATE LAC_CLANG_INCLUDE="${LIBCLANG_INCLUDE_DIR}")
find_program(CLANG_EXECUTABLE NAMES clang HINTS C:/msys64/ucrt64/bin REQUIRED)
execute_process(COMMAND ${CLANG_EXECUTABLE} -print-resource-dir OUTPUT_VARIABLE CLANG_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
target_compile_definitions(lac PRIVATE LAC_CLANG_RESOURCE="${CLANG_RESOURCE_DIR}")
target_compile_definitions(lac PRIVATE LAC_RUNTIME_PATH="${CMAKE_CURRENT_SOURCE_DIR}/runtime/lac_runtime.h")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lac PRIVATE -Wall -Wextra -Wpedantic)
endif()
